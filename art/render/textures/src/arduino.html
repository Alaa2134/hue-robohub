// RoboHub Bootcamp · Week 3 — Motors & Drivers
const int ENA = 25, IN1 = 26, IN2 = 27;

void setup() {
  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  ledcAttach(ENA, 20000, 10);   // 20 kHz, 10-bit PWM
  Serial.begin(115200);
}

void loop() {
  for (int duty = 0; duty <= 1023; duty += 16) {
    digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
    ledcWrite(ENA, duty);
    Serial.printf("duty=%d rpm=%d\n", duty, readRpm());
    delay(40);
  }
}
