1// HUE RoboHub — line follower control loop (1 kHz)
2#include "pid.hpp"
3#include "sensors.hpp"
4
5constexpr float KP = 1.85f, KI = 0.42f, KD = 0.091f;
6static Pid steer(KP, KI, KD, /*limit*/ 1.0f);
7
8void IRAM_ATTR control_tick() {
9  const float err = sensors.line_position();   // −1 … +1
10  const float u   = steer.update(err, DT);
11  const float base = adaptive_speed(err, curvature);
12
13  motors.set(base * (1.0f - u), base * (1.0f + u));
14  telemetry.push({err, u, base, encoders.rpm()});
15}
16
17float adaptive_speed(float err, float k) {
18  // slow into corners, full power on straights
19  return clamp(V_MAX * (1.0f - 0.65f * fabsf(k)), V_MIN, V_MAX);
20}
